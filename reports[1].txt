<script>
const token = localStorage.getItem('mm_token');
if (!token) window.location.href = '/login.html';

let allSignals = [], filteredSignals = [], period = 'all', page = 1;
const PAGE_SIZE = 15;

// ── Load ──────────────────────────────────────
async function loadData() {
  try {
    const r = await fetch('/api/signals', { headers: {'x-auth-token': token} });
    if (!r.ok) { logout(); return; }
    allSignals = await r.json();
    populateStratFilter();
    applyPeriod();
  } catch(e) {}
}
loadData();

// ── Period ────────────────────────────────────
function setPeriod(p, btn) {
  period = p;
  document.querySelectorAll('.period-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  applyPeriod();
}

function applyPeriod() {
  const now = new Date();
  let from = null;
  const labels = {all:'كل الإشارات',today:'إشارات اليوم',week:'هذا الأسبوع',month:'هذا الشهر','3months':'آخر 3 أشهر'};
  document.getElementById('period-label').textContent = labels[period] || '';

  if (period === 'today') { from = new Date(now.getFullYear(), now.getMonth(), now.getDate()); }
  else if (period === 'week') { from = new Date(now - 7*86400000); }
  else if (period === 'month') { from = new Date(now.getFullYear(), now.getMonth(), 1); }
  else if (period === '3months') { from = new Date(now - 90*86400000); }

  filteredSignals = from ? allSignals.filter(s => new Date(s.receivedAt) >= from) : [...allSignals];
  page = 1;
  updateAll();
}

function updateAll() {
  updateSummary();
  updateDonut();
  updateBarsChart();
  updateDailyChart();
  renderTable();
}

// ── Summary ───────────────────────────────────
function updateSummary() {
  const open    = filteredSignals.filter(s=>s.status==='open').length;
  const target  = filteredSignals.filter(s=>s.status==='target').length;
  const stopped = filteredSignals.filter(s=>s.status==='stopped').length;
  const closed  = target + stopped;
  const wr      = closed > 0 ? (target/closed*100).toFixed(1) : null;

  // متوسط نسبة الربح للصفقات الرابحة
  const winSigs = filteredSignals.filter(s=>s.status==='target');
  let avgProfit = '—';
  if (winSigs.length > 0) {
    const profits = winSigs.map(s => {
      const entry = parseFloat(s.price)||0;
      let tp = parseFloat(s.tp)||0;
      const sl = parseFloat(s.sl)||0;
      if(!tp||Math.abs(tp-entry)<0.00001) tp = s.action==='BUY'?entry+Math.abs(entry-sl)*2:entry-Math.abs(entry-sl)*2;
      return entry>0?(s.action==='BUY'?(tp-entry)/entry*100:(entry-tp)/entry*100):0;
    }).filter(p=>p>0);
    if (profits.length > 0) avgProfit = '+' + (profits.reduce((a,b)=>a+b,0)/profits.length).toFixed(2) + '%';
  }

  document.getElementById('s-total').textContent   = filteredSignals.length;
  document.getElementById('s-win').textContent     = target;
  document.getElementById('s-loss').textContent    = stopped;
  document.getElementById('s-open').textContent    = open;
  document.getElementById('s-wr').textContent      = wr ? wr+'%' : '—';
  document.getElementById('s-win-rate').textContent = closed>0?`${(target/closed*100).toFixed(1)}% من المغلقة`:'—';
  document.getElementById('s-loss-rate').textContent= closed>0?`${(stopped/closed*100).toFixed(1)}% من المغلقة`:'—';
  document.getElementById('s-avg-profit').textContent = avgProfit;
}

// ── Donut Chart ───────────────────────────────
function updateDonut() {
  const target  = filteredSignals.filter(s=>s.status==='target').length;
  const stopped = filteredSignals.filter(s=>s.status==='stopped').length;
  const open    = filteredSignals.filter(s=>s.status==='open').length;
  const closed  = target + stopped;
  const wr      = closed > 0 ? target/closed : 0;
  const circ    = 2 * Math.PI * 50; // ~314

  const winArc  = wr * circ;
  const lossArc = (1-wr) * circ;

  document.getElementById('donut-win').setAttribute('stroke-dasharray',  `${winArc} ${circ-winArc}`);
  document.getElementById('donut-loss').setAttribute('stroke-dasharray', `${lossArc} ${circ-lossArc}`);
  document.getElementById('donut-loss').setAttribute('stroke-dashoffset', `-${winArc}`);
  document.getElementById('donut-pct').textContent  = closed>0?(wr*100).toFixed(1)+'%':'—';
  document.getElementById('donut-pct').style.color  = wr>=0.6?'var(--green)':wr>=0.4?'var(--yellow)':'var(--red)';
  document.getElementById('leg-win').textContent    = target;
  document.getElementById('leg-loss').textContent   = stopped;
  document.getElementById('leg-open').textContent   = open;

  // أفضل استراتيجية
  const sm = {};
  filteredSignals.filter(s=>s.status==='target').forEach(s=>{sm[s.strategy]=(sm[s.strategy]||0)+1;});
  const best = Object.keys(sm).sort((a,b)=>sm[b]-sm[a])[0]||'—';
  document.getElementById('best-strat').textContent = best;
}

// ── Bars Chart (Best Symbols) ─────────────────
function updateBarsChart() {
  const symMap = {};
  filteredSignals.forEach(s => {
    const k = s.symbol.toUpperCase();
    if (!symMap[k]) symMap[k] = {win:0,loss:0,total:0};
    symMap[k].total++;
    if (s.status==='target')  symMap[k].win++;
    if (s.status==='stopped') symMap[k].loss++;
  });

  const sorted = Object.entries(symMap)
    .filter(([,v])=>v.win+v.loss>0)
    .sort((a,b)=>{
      const ra = a[1].win/(a[1].win+a[1].loss);
      const rb = b[1].win/(b[1].win+b[1].loss);
      return rb-ra;
    }).slice(0,8);

  document.getElementById('bars-count').textContent = sorted.length > 0 ? `(${sorted.length} سهم)` : '';

  if (!sorted.length) {
    document.getElementById('bars-chart').innerHTML = '<div style="text-align:center;padding:30px;color:var(--muted);font-size:12px">لا توجد صفقات مغلقة</div>';
    return;
  }

  const html = sorted.map(([sym, data]) => {
    const closed = data.win + data.loss;
    const wr = closed > 0 ? data.win/closed : 0;
    const pct = (wr*100).toFixed(0);
    const clr = wr>=0.6?'var(--green)':wr>=0.4?'var(--yellow)':'var(--red)';
    return `<div class="bar-item">
      <span class="bar-sym">${sym.replace('USDT','')}</span>
      <div class="bar-track">
        <div class="bar-fill" style="width:${pct}%;background:${clr}">
          <span class="bar-val">${pct}%</span>
        </div>
      </div>
      <span class="bar-count" style="color:${clr}">${data.win}/${closed}</span>
    </div>`;
  }).join('');

  document.getElementById('bars-chart').innerHTML = html;
}

// ── Daily Chart ───────────────────────────────
function updateDailyChart() {
  const days = 14;
  const buckets = [];
  const now = new Date();

  for (let i = days-1; i >= 0; i--) {
    const d = new Date(now - i*86400000);
    const key = d.toISOString().slice(0,10);
    buckets.push({key, date:d, win:0, loss:0});
  }

  filteredSignals.forEach(s => {
    if (!s.receivedAt) return;
    const key = new Date(s.receivedAt).toISOString().slice(0,10);
    const b = buckets.find(b=>b.key===key);
    if (!b) return;
    if (s.status==='target')  b.win++;
    if (s.status==='stopped') b.loss++;
  });

  const maxVal = Math.max(...buckets.map(b=>b.win+b.loss), 1);

  const chartEl  = document.getElementById('daily-chart');
  const labelsEl = document.getElementById('day-labels');

  chartEl.innerHTML = buckets.map(b => {
    const total = b.win + b.loss;
    const wh = total > 0 ? Math.max(4, (b.win/maxVal)*70) : 0;
    const lh = total > 0 ? Math.max(4, (b.loss/maxVal)*70) : 0;
    return `<div class="day-bar" title="${b.key}: ${b.win} رابح، ${b.loss} خاسر">
      ${b.win>0  ? `<div class="day-win"  style="height:${wh}px"  title="${b.win} رابح"></div>`  : ''}
      ${b.loss>0 ? `<div class="day-loss" style="height:${lh}px" title="${b.loss} خاسر"></div>` : ''}
      ${total===0 ? '<div style="height:2px;width:100%;background:var(--bg3);border-radius:1px"></div>' : ''}
    </div>`;
  }).join('');

  labelsEl.innerHTML = buckets.map(b => {
    const d = b.date;
    return `<div class="day-lbl">${d.getDate()}/${d.getMonth()+1}</div>`;
  }).join('');
}

function renderTable() {
  const search  = document.getElementById('filter-search').value.toLowerCase();
  const status  = document.getElementById('filter-status').value;
  const action  = document.getElementById('filter-action').value;
  const strat   = document.getElementById('filter-strat').value;

  let list = [...filteredSignals];
  if (search)         list = list.filter(s=>s.symbol.toLowerCase().includes(search));
  if (status!=='all') list = list.filter(s=>s.status===status);
  if (action!=='all') list = list.filter(s=>s.action===action);
  if (strat!=='all')  list = list.filter(s=>s.strategy===strat);

  const total = list.length;
  const pages = Math.ceil(total/PAGE_SIZE);
  if (page > pages) page = 1;
  const start = (page-1)*PAGE_SIZE;
  const pageList = list.slice(start, start+PAGE_SIZE);

  document.getElementById('table-count').textContent = `${total} صفقة`;

  const tbody = document.getElementById('history-tbody');
  if (!pageList.length) {
    tbody.innerHTML='<tr class="empty-row"><td colspan="10">لا توجد نتائج</td></tr>';
  } else {
    tbody.innerHTML = pageList.map((s,i) => {
      const entry = parseFloat(s.price)||0;
      let tp = parseFloat(s.tp)||0;
      const sl = parseFloat(s.sl)||0;
      const slDist = Math.abs(entry-sl);
      if(!tp||Math.abs(tp-entry)<0.00001) tp = s.action==='BUY'?entry+slDist*2:entry-slDist*2;
      const isBuy = s.action==='BUY';
      const tpPct = entry>0?(isBuy?(tp-entry)/entry*100:(entry-tp)/entry*100).toFixed(2):'—';
      const dp = entry>=100?2:4;
      const stMap = {open:{cls:'s-open',txt:'🟢 مفتوحة'},stopped:{cls:'s-stopped',txt:'🛑 وقف'},target:{cls:'s-target',txt:'🎯 هدف'}};
      const st = stMap[s.status]||stMap.open;
      const dt = new Date(s.receivedAt).toLocaleString('ar',{year:'numeric',month:'short',day:'numeric',hour:'2-digit',minute:'2-digit',hour12:true});
      return `<tr>
        <td style="color:var(--muted2);font-family:var(--mono);font-size:10px">${start+i+1}</td>
        <td style="font-family:var(--mono);font-size:12px;font-weight:500">${s.symbol}</td>
        <td><span style="font-family:var(--mono);font-size:11px;font-weight:600" class="${isBuy?'up':'dn'}">${isBuy?'▲ شراء':'▼ بيع'}</span></td>
        <td style="font-family:var(--mono);font-size:11px">$${entry.toFixed(dp)}</td>
        <td style="font-family:var(--mono);font-size:11px;color:var(--green)">$${tp.toFixed(dp)}</td>
        <td style="font-family:var(--mono);font-size:11px;color:var(--red)">$${sl.toFixed(dp)}</td>
        <td style="font-family:var(--mono);font-size:11px;color:var(--yellow)">+${tpPct}%</td>
        <td><span class="status-chip ${st.cls}">${st.txt}</span></td>
        <td style="font-size:11px;color:var(--muted2)">${s.strategy||'—'}</td>
        <td style="font-family:var(--mono);font-size:10px;color:var(--muted);white-space:nowrap">${dt}</td>
      </tr>`;
    }).join('');
  }

  // Pagination
  const pagEl = document.getElementById('pagination');
  if (pages <= 1) { pagEl.innerHTML=''; return; }
  let pagHtml = `<button class="page-btn" onclick="goPage(${page-1})" ${page===1?'disabled':''}>◀</button>`;
  for (let p=1;p<=pages;p++) {
    if(p===1||p===pages||Math.abs(p-page)<=1) pagHtml+=`<button class="page-btn ${p===page?'active':''}" onclick="goPage(${p})">${p}</button>`;
    else if(Math.abs(p-page)===2) pagHtml+='<span class="page-info">...</span>';
  }
  pagHtml+=`<button class="page-btn" onclick="goPage(${page+1})" ${page===pages?'disabled':''}>▶</button>`;
  pagHtml+=`<span class="page-info">${start+1}-${Math.min(start+PAGE_SIZE,total)} من ${total}</span>`;
  pagEl.innerHTML=pagHtml;
}

function goPage(p){
  page=p;
  renderTable();
  document.querySelector('.history-box').scrollIntoView({behavior:'smooth'});
}
